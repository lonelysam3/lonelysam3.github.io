{{- /* ========================================================================== lang-switch.html — 语言切换器 -------------------------------------------------------------------------- 行为：点击语言按钮 → 跳转到【同一篇文章】在目标语言下的版本。 【三种情况的处理】 1. 该文章有译文 → 链接到译文 2. 该文章是文章页但无译文 → 不做成链接（不把读者送去 404）， 改为灰色不可点的标记，作者一看就知道该补一篇 3. 非文章页（首页/列表/静态页）→ 退回到目标语言的首页 【为什么不能用 .AllTranslations 判断「有没有译文」】 .AllTranslations 只列出【已经存在】的译文，所以「列表为空」既可能是 「这篇没翻译」，也可能是「别的语言根本没有这篇」。要区分这两者，必须 遍历每种语言的站点、在其中按 TranslationKey 查找——下面用的就是这种方式 （site.Sites 给出全部语言站点，比拿 site.Home.AllTranslations 反查更直接）。 样式见 30-header.css 的 .lang-switch。 ========================================================================== */ -}} {{- if gt (len site.Languages) 1 }} {{- $cur := site.Language.Lang }}
{{- range site.Languages }} {{- $code := .Lang }} {{- /* 按钮上只用【短标签】：中文站显示 "EN"、英文站显示 "中"。 不用 "English" / "中文" 这类全称——两个全称并排会让切换器宽出近一倍， 实测会盖住导航最后一项（英文站 "Stats" 被遮掉一半）。 短标签也符合这类控件的惯例（GitHub、MDN 都用 EN / 中 这种写法）。 */ -}} {{- $label := upper $code }} {{- if eq $code "zh" }}{{ $label = "中" }}{{ end }} {{- if eq $code $cur }} {{- /* 当前语言：标记为选中，不渲染成链接 */ -}} {{ $label }} {{- else }} {{- /* ---- 在目标语言站点里按 TranslationKey 找这篇的译文 ---- */ -}} {{- $url := "" }} {{- range site.Sites }} {{- if eq .Language.Lang $code }} {{- range .RegularPages }} {{- if eq .TranslationKey $.TranslationKey }}{{ $url = .RelPermalink }}{{ end }} {{- end }} {{- end }} {{- end }} {{- if $url }} {{ $label }} {{- else if and $.IsPage (ne $.Section "") }} {{- /* 文章页且尚无译文：不做成链接，避免把读者送到 404 */ -}} {{ $label }} {{- else }} {{- /* 非文章页：退回到该语言首页。URL 取自 site.Home.AllTranslations， 由 Hugo 给出——它已处理好 defaultContentLanguageInSubdir （本站默认语言不带前缀、英文带 /en/）。 */ -}} {{- $home := "" }} {{- range site.Home.AllTranslations }} {{- if eq .Language.Lang $code }}{{ $home = .RelPermalink }}{{ end }} {{- end }} {{ $label }} {{- end }} {{- end }} {{- end }}
{{- end }}
