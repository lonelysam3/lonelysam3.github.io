{{- /* ========================================================================== callout.html — 提示框短代码 -------------------------------------------------------------------------- 用法： {{< callout type="note" title="可选标题" >}} 支持 **Markdown** 内容。 {{< /callout >}} type: note | tip | warn | danger | info （默认 note） title: 可选，省略时按 type 取默认中文标题 实现要点： · 内容用 .Inner | markdownify，因此框内可以正常写 Markdown（列表、行内 代码、链接都行），不像原样输出那样只能写纯文本。 · .InnerDeindent 去掉缩进，避免短代码内的缩进被 Markdown 当成代码块。 · 类型只影响 data-type，具体配色由 CSS 负责（见 60-components.css）， 这样以后调色不需要改模板。 ========================================================================== */ -}} {{- $type := .Get "type" | default "note" -}} {{- /* 【默认标题走 i18n，不再写死中文表】 原先这里是一份硬编码的中文 dict（说明/提示/注意/警告/信息）， 英文站的提示框标题因此永远显示中文。i18n 里早已备好 callout_note / callout_tip / callout_warn / callout_danger / callout_info 五个键，只是这里从没查过。 键名按 type 拼出来，所以未知 type 会取到空串 —— 回退到 note。 */ -}} {{- $title := .Get "title" -}} {{- if not $title -}} {{- $title = i18n (printf "callout_%s" $type) -}} {{- if not $title -}} {{- $title = i18n "callout_note" -}} {{- end -}} {{- end -}}
